# Paths (shell globs, relative to the partition root, no leading /) that may hold secrets. idea#322.
# Their CONTENT never goes into a manifest. Known formats are normalised first and the normalised text is
# hashed (so a change in the non-secret part is still caught); anything else is recorded as "secret".
#   norm:shadow   crypt hash field -> <hash>      (etc/shadow, etc/gshadow)
#   norm:keyval   values of psk=, password=, wep-key*=, private-key-password=, psk "..." -> <secret>
# Format: <glob>  [norm:<name>]
etc/shadow                                   norm:shadow
etc/gshadow                                  norm:shadow
etc/NetworkManager/system-connections/*      norm:keyval
etc/wpa_supplicant/*.conf                    norm:keyval
etc/config/wireless                          norm:keyval
etc/idea/secrets*
etc/ssh/ssh_host_*
root/.ssh/*
home/*/.ssh/*
etc/ssl/private/*
etc/tailscale/*
var/lib/tailscale/*
etc/dropbear/*
